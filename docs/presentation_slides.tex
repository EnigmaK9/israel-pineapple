% SLIDE 5: RELATIONAL DATA
% ---------------------------------------------------------
\begin{frame}{4. Relational Data Architecture (SQLite)}
  \begin{itemize}\setlength{\itemsep}{0.8em}
    \item Normalized schema in Third Normal Form.
    \item Table \texttt{lineas} stores machine asset metadata.
    \item Table \texttt{produccion\_scada} stores time-series telemetry.
    \item Foreign key constraints enforce relational integrity.
    \item Parameterized queries eliminate SQL injection vulnerabilities.
  \end{itemize}
\end{frame}
